# Q7 – REST JSON API (GET, POST, PUT, DELETE)

`public/api/products.php`:
```php
<?php
header('Content-Type: application/json');
require_once __DIR__ . '/../../config/database.php';
$db = Database::connect();
$data = json_decode(file_get_contents('php://input'), true) ?? [];
$method = $_SERVER['REQUEST_METHOD'];

try {
    if ($method === 'GET') {
        echo json_encode($db->query('SELECT * FROM products')->fetchAll()); exit;
    }
    if ($method === 'POST') {
        $name = trim($data['name'] ?? '');
        $price = filter_var($data['price'] ?? null, FILTER_VALIDATE_FLOAT);
        if ($name === '' || $price === false || $price < 0) {
            http_response_code(422); echo json_encode(['error'=>'Invalid data']); exit;
        }
        $s=$db->prepare('INSERT INTO products(name,price) VALUES(?,?)');
        $s->execute([$name,$price]);
        http_response_code(201); echo json_encode(['message'=>'Added']); exit;
    }
    $id = filter_var($data['id'] ?? $_GET['id'] ?? null, FILTER_VALIDATE_INT);
    if (!$id || $id < 1) { http_response_code(400); echo json_encode(['error'=>'Invalid id']); exit; }
    if ($method === 'PUT') {
        $name=trim($data['name'] ?? '');
        $price=filter_var($data['price'] ?? null,FILTER_VALIDATE_FLOAT);
        if ($name==='' || $price===false || $price<0) {
            http_response_code(422); echo json_encode(['error'=>'Invalid data']); exit;
        }
        $s=$db->prepare('UPDATE products SET name=?,price=? WHERE id=?');
        $s->execute([$name,$price,$id]); echo json_encode(['message'=>'Updated']); exit;
    }
    if ($method === 'DELETE') {
        $s=$db->prepare('DELETE FROM products WHERE id=?');
        $s->execute([$id]); echo json_encode(['message'=>'Deleted']); exit;
    }
    header('Allow: GET, POST, PUT, DELETE');
    http_response_code(405); echo json_encode(['error'=>'Method not allowed']);
} catch (Throwable $e) {
    error_log($e->getMessage()); http_response_code(500);
    echo json_encode(['error'=>'Server error']);
}
```
